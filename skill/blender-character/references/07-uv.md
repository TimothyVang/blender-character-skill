# 07 UV unwrapping

Course source: chapter 10 "UV Unwrapping Your Character", 10:44:02 to 12:05:29. The video uses Blender 5.0.1; the
snippets here were run in Blender 4.5.16.

## Goal

Every low-poly (retopologised) object has one UV map with seams hidden where the camera and the clothes will not show
them, no visible stretching, no overlapping islands, and all islands inside the single 0 to 1 square. The face has more
texture area than the rest of the body; parts that will be covered (feet inside shoes, the inside of garments) have
less. No UDIMs are used.

## Starting state

- The retopologised meshes from stage 06, named as in SKILL.md: `body_retopo`, `Shirt_retopo`, `Skirt_retopo`,
  `Boot.L_retopo` and `Boot.R_retopo` (plus any separate boot parts, named `Boot.L_<part>_retopo`), `Eye.L`, `Eye.R`
  and `Mouth` (teeth and tongue). The course's own screen names are lower case (`shirt_retopo`, `eye`, `jaw`); it
  names everything before starting [10:50:13], and every snippet below takes the object by name.
- Symmetric objects really are symmetric, and **the object origin lies on the mirror plane** (`select_mirror` works
  in the object's own space). If mirroring a selection fails, do what the course does, `Mesh > Snap to Symmetry`
  [10:53:20]: `bpy.ops.mesh.symmetry_snap(direction='NEGATIVE_X', threshold=0.05, factor=0.5, use_center=True)`.
- Topology has clean edge loops at wrists, ankles, neck, finger bases, the lip line, the eyelid rim and round the
  ears. The course relies on this: every seam is a loop or a short path along existing edges [10:56:19].
- The high-poly sculpts are NOT unwrapped [12:03:55]. Leave them alone.

## Order of work

The course's checklist [10:49:48] is three passes per object: cut seams, fix stretching, arrange islands. Body
first, then each garment, then the shoes together, then the downloaded eye and teeth meshes. The course turns on
Live Unwrap in both editors [10:51:37] so the map updates after every seam and every moved pin. You do not have
that loop; call `unwrap()` after each batch of seams. If its stdout says "Unwrap failed to solve N island(s)", a
closed shape still has no seam. Load the helpers once per session:

```python
# verified in 4.5.16 background
import bpy, bmesh, math, numpy as np
from mathutils import Vector

def edit(obj):
    """Put obj alone into Edit Mode, edge select. Returns its BMesh."""
    if bpy.context.object and bpy.context.object.mode != 'OBJECT':
        bpy.ops.object.mode_set(mode='OBJECT')
    bpy.ops.object.select_all(action='DESELECT')
    obj.select_set(True); bpy.context.view_layer.objects.active = obj
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.context.tool_settings.mesh_select_mode = (False, True, False)
    return bmesh.from_edit_mesh(obj.data)

def edit_many(objs):
    """Put several objects into Edit Mode together (they then share one UV square)."""
    edit(objs[0])
    for o in objs: o.select_set(True)
    bpy.ops.object.mode_set(mode='OBJECT'); bpy.ops.object.mode_set(mode='EDIT')

def _mark(obj, mirror_x):
    bpy.context.tool_settings.mesh_select_mode = (False, True, False)
    if mirror_x:
        bpy.ops.mesh.select_mirror(axis={'X'}, extend=True)
    bpy.ops.mesh.mark_seam(clear=False)
    return sum(e.select for e in bmesh.from_edit_mesh(obj.data).edges)   # edges just marked

def seam_loop(obj, point, direction, mirror_x=True):
    """Seam on the whole edge loop passing nearest world `point` and running along world `direction`."""
    bm = edit(obj); bpy.ops.mesh.select_all(action='DESELECT')
    inv = obj.matrix_world.inverted()
    p = inv @ Vector(point); d = (inv.to_3x3() @ Vector(direction)).normalized()
    def score(e):
        mid = (e.verts[0].co + e.verts[1].co) / 2
        return (mid - p).length * (2.0 - abs((e.verts[1].co - e.verts[0].co).normalized().dot(d)))
    e = min(bm.edges, key=score); e.select_set(True)
    bm.select_history.clear(); bm.select_history.add(e); bmesh.update_edit_mesh(obj.data)
    bpy.ops.mesh.loop_multi_select(ring=False)
    return _mark(obj, mirror_x)

def seam_path(obj, point_a, point_b, mirror_x=True):
    """Seam along the shortest edge path between the vertices nearest two world points."""
    bm = edit(obj); bpy.ops.mesh.select_all(action='DESELECT')
    bpy.context.tool_settings.mesh_select_mode = (True, False, False)
    inv = obj.matrix_world.inverted()
    a, b = (min(bm.verts, key=lambda v: (v.co - inv @ Vector(q)).length_squared) for q in (point_a, point_b))
    a.select = b.select = True
    bm.select_history.clear(); bm.select_history.add(a); bm.select_history.add(b); bmesh.update_edit_mesh(obj.data)
    bpy.ops.mesh.shortest_path_select(edge_mode='SELECT')
    return _mark(obj, mirror_x)

def seam_border(obj, face_test, mirror_x=False):
    """Seam round the border of a face region. face_test(world_centre) -> True for faces inside the region."""
    bm = edit(obj); bpy.ops.mesh.select_all(action='DESELECT')
    bpy.context.tool_settings.mesh_select_mode = (False, False, True)
    for f in bm.faces:
        f.select = bool(face_test(obj.matrix_world @ f.calc_center_median()))
    bmesh.update_edit_mesh(obj.data)
    bpy.ops.mesh.region_to_loop()
    return _mark(obj, mirror_x)

def unwrap(objs, method='MINIMUM_STRETCH'):
    """Unwrap all faces of one or several objects together and give every island the same texel density."""
    edit_many(objs); bpy.ops.mesh.select_all(action='SELECT')
    bpy.ops.uv.unwrap(method=method, fill_holes=True, correct_aspect=True, margin=0.001, iterations=10)
    bpy.ops.uv.select_all(action='SELECT'); bpy.ops.uv.average_islands_scale()
    bpy.ops.object.mode_set(mode='OBJECT')

def pack(objs, margin=0.01, rotate_method='CARDINAL'):
    """Pack every island of the objects into the one 0..1 square. Relative island sizes are kept."""
    edit_many(objs); bpy.ops.mesh.select_all(action='SELECT'); bpy.ops.uv.select_all(action='SELECT')
    bpy.ops.uv.pack_islands(rotate=True, rotate_method=rotate_method, scale=True, margin_method='FRACTION',
                            margin=margin, shape_method='CONCAVE')
    bpy.ops.object.mode_set(mode='OBJECT')
```

### 1. Body: cut the limbs and neck as tubes [10:51:04 to 10:56:10]

Why: an arm, a leg and the neck are cylinders. A cylinder lies flat once it has a ring cut at each end and one
cut down its length. Where the lengthwise cut goes is the real decision; the rule is "where it is not seen"
[10:48:01]:

| Part | Ring seams | Lengthwise seam |
|---|---|---|
| Leg | top of the leg, and the ankle | down the inner leg |
| Arm | where the arm meets the torso, chosen on a loop tucked into the armpit side and not out on the shoulder; and the wrist | along the underside of the arm |
| Neck | the base (the shirt covers it) and a loop high up under the jaw, chosen because a loop lower on the neck is seen from the front | down the centre back [qy-wwP-b9HY-10-105606] 10:56:11 |

### 4. Look for stretching [11:04:25 to 11:11:45]

The course uses two views. In the UV editor, `Overlays > Display Stretch`: blue is good, yellow or red is
stretched [11:05:15]. On the model, a material `uv checker` whose Base Color is a generated image `uv grid` (Color
Grid, 4096 x 4096) [11:05:57], seen in Solid shading with `Color > Texture`. Squares of unequal size or slanted
letters mean stretch; the chest, feet, fingers, ears, nose and lips showed it [11:07:14].

The fix is the unwrap method. Of Angle Based, Conformal and Minimum Stretch the course uses **Minimum Stretch**
on each bad island (Conformal is called fit only for simple objects), then `UV > Average Islands Scale` so all
islands have equal-size squares again [11:08:09 to 11:09:40]. `unwrap()` above does both for every island.

```python
# verified in 4.5.16 background
def checker(objs, size=4096):
    """Give objects the course's checker material so stretching can be seen in a capture."""
    img = bpy.data.images.get("uv grid") or bpy.data.images.new("uv grid", size, size); img.generated_type = 'COLOR_GRID'
    mat = bpy.data.materials.get("uv checker")
    if mat is None:
        mat = bpy.data.materials.new("uv checker"); mat.use_nodes = True
        tex = mat.node_tree.nodes.new("ShaderNodeTexImage"); tex.image = img
        mat.node_tree.links.new(tex.outputs["Color"], mat.node_tree.nodes["Principled BSDF"].inputs["Base Color"])
    for o in objs:
        if mat.name not in o.data.materials: o.data.materials.append(mat)
        o.active_material_index = o.data.materials.find(mat.name)
    return mat
checker([body])
```

### 5. Hand-fix the face with pins [11:11:45 to 11:22:50]

Why: Minimum Stretch still leaves the lips, nose and eyelids uneven, and these are where a viewer looks.
How the course does it: move the face island into free space, then pin UVs (`P`). With live unwrap
on, moving a pinned UV drags the unpinned ones along, so a few pins steer the island. The outer border, the eye
rims and the ear holes are pinned first so the good parts cannot move. Then:

- Mouth: the lip rows are scaled together in V (`S Y`) to close the gap, then rows are nudged with `G Y` and the
  UV Grab sculpt tool until the checker squares on the lips are even [11:15:40 to 11:18:40].
- Nose: its UVs are pinned and pushed outward with the Grab tool until its squares match the cheeks [11:19:40].
- Eyelids: the eye-opening border of the face island is scaled to almost nothing in V, a flat slit [11:21:40].
  Reason given: the eyes will close; laid out like this the lid texture is not stretched when the eye is shut.
- Clear every pin when done (`Alt P`). Left-over pins break later unwraps, also on the hair scalps of stage 08.

You have no UV sculpt brush. "Move a few UVs, pin them, let the rest relax round them" scripts well:

```python
# verified in 4.5.16 background
def flatten_uv(obj, vert_test, factor=0.0, axis=None, pin=True, relax=200):
    """Scale the UVs of the vertices picked by vert_test(world_co) towards their mean on one UV axis (0 = U,
    1 = V; default: the axis they spread least on). factor 0 = a straight line. Pins them and re-solves the rest."""
    bm = edit(obj); uv = bm.loops.layers.uv.verify(); mw = obj.matrix_world
    loops = [l for f in bm.faces for l in f.loops if vert_test(mw @ l.vert.co)]
    if not loops: return 0
    if axis is None:
        spread = [max(l[uv].uv[k] for l in loops) - min(l[uv].uv[k] for l in loops) for k in (0, 1)]
        axis = 0 if spread[0] < spread[1] else 1
    mean = sum(l[uv].uv[axis] for l in loops) / len(loops)
    for l in loops:
        l[uv].uv[axis] = mean + (l[uv].uv[axis] - mean) * factor
        l[uv].pin_uv = pin
    bmesh.update_edit_mesh(obj.data)
    bpy.ops.mesh.select_all(action='SELECT'); bpy.ops.uv.select_all(action='SELECT')
    bpy.ops.uv.minimize_stretch(fill_holes=True, blend=0.0, iterations=relax)   # relaxes from the current layout;
    return len(loops)                                                           # pinned UVs stay where they are

def clear_pins(obj):
    edit(obj); bpy.ops.mesh.select_all(action='SELECT'); bpy.ops.uv.select_all(action='SELECT')
    bpy.ops.uv.pin(clear=True); bpy.ops.object.mode_set(mode='OBJECT')

n = flatten_uv(body, lambda p: abs(p.z - 0.41) < 0.005 and p.x > 0, factor=0.2)   # example: one row of one island
clear_pins(body)
```

Pick vertices of ONE island that form a row; a seam vertex is moved in every island it belongs to. Run
`uv_report` after each call: if stretch went up or `flipped_faces` appeared, `clear_pins` and `unwrap([obj])`.

### 6. Arrange the islands [11:22:50 to 11:38:30]

The course packs once with `UV > Pack Islands` just to get everything into the square, then lays the islands out by
hand. What matters in that layout:

- **Scale is resolution.** Enlarging an island makes the checker letters on that part smaller, which means more
  pixels there; shrinking does the opposite [11:23:40]. So the face island is scaled up, and the ears are scaled up
  to match the face. The legs, the feet (they sit inside shoes), the soles and anything hidden are scaled down.
- No island overlaps another [11:25:55].
- Checker letters read upright and left-side parts sit on the left of the square, so a person can find things
  later [11:25:20]. This helps human painters only; it does not change the render.
- Empty space is accepted: the character is for animation, not a game, so the square is not squeezed [12:01:50].

Through bpy: set the relative sizes first, then let `pack()` place them. `pack_islands` keeps relative sizes.

```python
# verified in 4.5.16 background
def scale_islands(obj, face_test, factor):
    """Scale the UVs of whole islands picked by face_test(world_centre). Call after unwrap(), before pack()."""
    was_edit = obj.mode == 'EDIT'
    if was_edit: bpy.ops.object.mode_set(mode='OBJECT')
    bm = bmesh.new(); bm.from_mesh(obj.data); uv = bm.loops.layers.uv.verify(); mw = obj.matrix_world
    loops = [l for f in bm.faces if face_test(mw @ f.calc_center_median()) for l in f.loops]
    if loops:
        c = sum((l[uv].uv for l in loops), Vector((0, 0))) / len(loops)
        for l in loops: l[uv].uv = c + (l[uv].uv - c) * factor
    bm.to_mesh(obj.data); bm.free()
    if was_edit: bpy.ops.object.mode_set(mode='EDIT')
    return len(loops)

scale_islands(body, lambda c: c.z < 0.08, 0.6)      # example: everything below the ankle ring gets less area
pack([body])
```

`face_test` must pick complete islands (use the same boundaries as your seams) or the island tears.

### 8. Shoes: several objects, one square [11:49:10 to 12:01:55]

Why: a shoe is built from separate objects (upper, straps, sole). They are selected together and edited together

## How the artist works on screen

The course uses the UV Editing workspace: a UV editor beside the 3D viewport, normally in Edit Mode.
It orbits and zooms the model while comparing the selected geometry with its islands.
The highlighted edges and changing islands are more reliable than assuming a shortcut selected the intended loop.
These observations use the course's model; select the corresponding edges on your own topology.

## Mistakes the course warns about

- A seam where the camera sees it. Every lengthwise seam goes on an inner or under side, or under hair or cloth.
- Judging stretch from the UV editor only. The checker on the model is the real test [11:05:47].
- Marking seams with faces hidden and then unhiding: the map resets oddly; re-mark any seam to refresh [11:03:19].
- Scaling an island without remembering that it changes resolution there [11:24:09]; overlapping islands
  [11:26:05]; moving an island in vertex select with sync on, which drags its neighbours [11:27:51].
- Old pins: they stop later unwraps from working on that island [11:56:42].
- Forgetting an object. Go through every low-poly object with the checker before leaving [12:03:44].

## What changes for a different character

General: tubes get two rings and one hidden lengthwise cut; flat bodies of cloth split front and back; the insides
of the mouth and eye sockets are separate islands; Minimum Stretch then Average Islands Scale; scale by importance;
garments are cut where real garments are sewn.

Specific to this character: she has hair, so the scalp carries two long seams. For a bald character run the head
seam up the back of the skull only as far as needed and keep the top whole. Bare feet that are seen need normal
size. Long sleeves or trousers hide limb seams anywhere, so put them on the garment seam line.

- Creatures: tails, tentacles, horns and long snouts are tubes (ring at the root, cut along the underside). Wings
  and fins are front and back sheets cut along the leading edge. Put the body seam along the belly midline.
- Hard-surface parts: cut along sharp edges (`bpy.ops.mesh.edges_select_sharp(sharpness=0.52)` then mark seam),
  use `method='ANGLE_BASED'`, and `rotate_method='AXIS_ALIGNED'` in `pack()` so straight edges stay straight.
- Asymmetric characters: pass `mirror_x=False` and cut each side on its own [10:53:14].
- Games, or more detail than one 4K square holds: that is where UDIMs come in; the course leaves them out.

```python
# verified in 4.5.16 background
def uv_report(obj, texture_px=4096):
    """Stretch, island and bounds numbers for one object. Works from any mode."""
    was_edit = obj.mode == 'EDIT'
    if was_edit: bpy.ops.object.mode_set(mode='OBJECT')
    bm = bmesh.new(); bm.from_mesh(obj.data); uv = bm.loops.layers.uv.active
    if uv is None: bm.free(); return {"error": "no UV map"}
    mw = obj.matrix_world; parent = list(range(len(bm.faces))); first = {}
    def find(i):
        while parent[i] != i: parent[i] = parent[parent[i]]; i = parent[i]
        return i
    a3, a2, flipped, lo, hi = [], [], 0, [9, 9], [-9, -9]
    for f in bm.faces:
        co = [mw @ l.vert.co for l in f.loops]; uvs = [l[uv].uv for l in f.loops]; n = len(co)
        a3.append(sum((co[i] - co[0]).cross(co[i + 1] - co[0]).length for i in range(1, n - 1)) / 2)
        s = sum(uvs[i].x * uvs[(i + 1) % n].y - uvs[(i + 1) % n].x * uvs[i].y for i in range(n)) / 2
        a2.append(abs(s)); flipped += s < 0
        for l in f.loops:                         # faces sharing a vertex at the same UV are one island
            u = l[uv].uv; k = (l.vert.index, round(u.x, 5), round(u.y, 5))
            lo = [min(lo[0], u.x), min(lo[1], u.y)]; hi = [max(hi[0], u.x), max(hi[1], u.y)]
            if k in first: parent[find(f.index)] = find(first[k])
            else: first[k] = f.index
    a3, a2 = np.array(a3), np.array(a2); isl = np.array([find(i) for i in range(len(a3))])
    ok = (a3 > 1e-12) & (a2 > 1e-12); rows = []
    for i in np.unique(isl):
        m = (isl == i) & ok
        if not m.any(): continue
        r = (a2[m] / a2[m].sum()) / (a3[m] / a3[m].sum())             # 1.0 = face has its fair share of the island
        rows.append({"faces": int(m.sum()), "px_per_m": round(float(np.sqrt(a2[m].sum() / a3[m].sum()) * texture_px)),
                     "stretch_p95": round(float(np.percentile(np.maximum(r, 1 / r), 95)), 2)})
    bm.free()
    if was_edit: bpy.ops.object.mode_set(mode='EDIT')
    rows.sort(key=lambda d: -d["stretch_p95"])
    return {"islands": len(rows), "worst_islands": rows[:5], "zero_area_faces": int((~ok).sum()),
            "flipped_faces": int(flipped), "bounds": [round(v, 3) for v in lo + hi],
            "inside_0_1": bool(min(lo) >= 0 and max(hi) <= 1), "area_used": round(float(a2.sum()), 3),
            "px_per_m_min_max": [min(d["px_per_m"] for d in rows), max(d["px_per_m"] for d in rows)]}

def uv_overlap_faces(objs):
    """Number of faces whose UVs overlap another face's UVs, over all the objects together."""
    edit_many(objs); ts = bpy.context.tool_settings; old = ts.use_uv_select_sync
    ts.use_uv_select_sync = True; ts.mesh_select_mode = (False, False, True)
    bpy.ops.mesh.select_all(action='DESELECT'); bpy.ops.uv.select_overlap()
    n = sum(f.select for o in objs for f in bmesh.from_edit_mesh(o.data).faces)
    ts.use_uv_select_sync = old; bpy.ops.object.mode_set(mode='OBJECT')
    return n

result = {"report": uv_report(body), "overlap": uv_overlap_faces([body])}
```

Failed if any of these is true, per object:

- `error` is present, `zero_area_faces` > 0, `inside_0_1` is false, or `uv_overlap_faces` > 0.
- `flipped_faces` > 0 (an island is mirrored; text and normal maps would read backwards).
- An island's `stretch_p95` is above 2.0. This limit is mine; the course judges by eye. It means one face in twenty
  has twice or half its fair share of its island. Small fan islands (finger tips, the end of the mouth bag) may
  sit a little higher; look at them on the checker before accepting.
- `islands` is not what the seams should give (course body: 2 torso, 2 legs, 2 arms, neck, face, 4 ear halves,
  2 eye sockets, mouth bag, hand backs and palms, a pair per finger, foot tops and soles).
- The face island's `px_per_m` is not the highest on the body, or covered parts are not lower than visible ones.
  My suggested ratios, as the course gives none: face and ears 1.5 to 2 times the torso, hidden parts 0.5 times.
